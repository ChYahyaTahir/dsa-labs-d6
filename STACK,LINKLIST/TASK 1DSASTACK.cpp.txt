#include <iostream>
#include <stack>
using namespace std;

int prec(char c) {
    if (c == '+' || c == '-') return 1;
    if (c == '*' || c == '/') return 2;
    return 0;
}

int main() {
    int choice;
    cin >> choice;

    if (choice == 1) {
        string s;
        cin >> s;

        stack<char> st;
        for (int i = 0; i < s.length(); i++) {
            st.push(s[i]);
        }

        string rev = "";
        while (!st.empty()) {
            rev += st.top();
            st.pop();
        }

        cout << rev;
    }

    else if (choice == 2) {
        string s;
        cin >> s;

        stack<char> st;
        bool ok = true;

        for (int i = 0; i < s.length(); i++) {
            if (s[i] == '(') st.push(s[i]);
            else if (s[i] == ')') {
                if (st.empty()) {
                    ok = false;
                    break;
                }
                st.pop();
            }
        }

        if (!st.empty()) ok = false;

        if (ok) cout << "Balanced";
        else cout << "Not Balanced";
    }

    else if (choice == 3) {
        string s;
        cin >> s;

        stack<char> st;
        string res = "";

        for (int i = 0; i < s.length(); i++) {
            char c = s[i];

            if (isalnum(c)) res += c;

            else if (c == '(') st.push(c);

            else if (c == ')') {
                while (!st.empty() && st.top() != '(') {
                    res += st.top();
                    st.pop();
                }
                st.pop();
            }

            else {
                while (!st.empty() && prec(st.top()) >= prec(c)) {
                    res += st.top();
                    st.pop();
                }
                st.push(c);
            }
        }

        while (!st.empty()) {
            res += st.top();
            st.pop();
        }

        cout << res;
    }

    return 0;
}
