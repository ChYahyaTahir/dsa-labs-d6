#include <iostream>
#include <queue>
using namespace std;

int main() {
    queue<string> e, p, c;
    int ch;

    while (true) {
        cin >> ch;

        if (ch == 1) {
            string name;
            cin >> name;
            e.push(name);
        }

        else if (ch == 2) {
            string name;
            cin >> name;
            p.push(name);
        }

        else if (ch == 3) {
            string name;
            cin >> name;
            c.push(name);
        }

        else if (ch == 4) {
            if (!e.empty()) {
                cout << e.front() << endl;
                e.pop();
            }
            else if (!p.empty()) {
                cout << p.front() << endl;
                p.pop();
            }
            else if (!c.empty()) {
                cout << c.front() << endl;
                c.pop();
            }
            else {
                cout << "No Flights\n";
            }
        }

        else if (ch == 5) {
            cout << "Emergency: ";
            queue<string> t1 = e;
            while (!t1.empty()) {
                cout << t1.front() << " ";
                t1.pop();
            }
            cout << endl;

            cout << "Passenger: ";
            queue<string> t2 = p;
            while (!t2.empty()) {
                cout << t2.front() << " ";
                t2.pop();
            }
            cout << endl;

            cout << "Cargo: ";
            queue<string> t3 = c;
            while (!t3.empty()) {
                cout << t3.front() << " ";
                t3.pop();
            }
            cout << endl;
        }

        else if (ch == 6) break;
    }

    return 0;
}
